\section{Crossover designs}
\begin{frame}[fragile]{Crossover design }
A crossover design has the following components \citep{pat51}: 
\begin{block}{Components}
\begin{enumerate}
    \item  \textit{Sequences}: which are each one of the distinct treatment combinations to be sequentially and randomly applied to the experimental units.
    \item \textit{Treatments}: randomly applied to each experimental unit within each sequence.
    \item \textit{Periods}: the time intervals in which the treatments that make up a sequence are applied, usually, each period is the same for all sequences and, consequently, the number of periods is equal to the number of elements of each sequence.
    \item \textit{Experimental unit}: the subjects or elements that receive the treatments, the total number of experimental units in the study is denoted by  $n$.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}[fragile]{Carryover effects }
\begin{block}{Carryover effects}
 Persistency of the effect of a treatment over those that will be applied later, i.e., the treatment is applied in a given period, but its effect still affects the response in later periods when other treatments are applied. When the effect persists after one period, it is known as first order carryover effect, and when it persists after two periods, it is called a second order carry-over effect and so on \citep{vegas2016crossover}.
\end{block}
\begin{block}{Types}
\begin{enumerate}
    \item  \textbf{Simple carry-over}: where the residual effect of treatment affects equally each of the treatments that are preceded by it, and 
    \item \textbf{Complex carry-over}: where the residual effect of the treatment affects each of the other treatments differently.
\end{enumerate}
\end{block}
\end{frame}
\begin{frame}[fragile]{Background}
\vfill
\begin{enumerate}
    \item In \cite{curtin2017meta} and \cite{li2018power},  generalized linear models were used for crossover designs for two periods and two sequences of two treatments where the experimental unit was observed once in each period.
    \item \cite{josephy2015within}, \cite{hao2015explicit}, \cite{lui2015test},  \cite{grayling2018blinded}, \cite{madeyski2018effect} and \cite{kitchenham2018corrections} used mixed models for crossover designs with a normal response and a single observation per period, including additively carry-over effects in the model. 
    \item \cite{diaz2013random} and \cite{for15} normal mixed models are used to study crossover designs with repeated measures and without including carry-over effects.
    \item  Generalized estimating equations (GEE) were used by \cite{jankar2020optimal} to estimate model parameters in crossover experiments. 
\end{enumerate}
\end{frame}


\begin{frame}[fragile]{Outline}
 \begin{block}{Problems}
     \begin{itemize}
         \item There is no adequate methodology to perform a deviance analysis table with carryover effects.
         \item It is necessary to build an inference that performs well in scenarios of a few experimental units.
         \item The usual correlation structures are very restrictive if the number of observations per period is large.
         \item Assuming that the carry-over effects are constant over time is very naive.
     \end{itemize}
 \end{block}    
 \begin{block}{Solutions}
     \begin{enumerate}
         \item  A Deviance Analysis methodology was constructed.
         \item A new correlation structure using the Kronecker product was developed.
         \item A semiparametric GEE model was developed to estimate the temporal and carry-over effects.
         %% Buscar sinonimos
     \end{enumerate}
 \end{block}
\end{frame}

\begin{frame}[fragile]{Notation}
 In a crossover design with $S$ sequences of length (the number of elements comprising each sequence)  $P$, $Y_{ijk}$ is defined as the  $k$th response of the $i$th experimental unit at $j$th period. Let $n_{ij}$ be the number of observations of the $i$th experimental unit during the  $j$th period. Then, vector $\pmb{Y}_{ij}$ is defined as: 
 \begin{equation}\nonumber
  \pmb{Y}_{ij}=\left(Y_{ij1}, \ldots, Y_{ijn_{ij}} \right)^T
 \end{equation}
Also, vector $\pmb{Y}_{i}$ is defined as:
 \begin{equation}\nonumber
\pmb{Y}_{i}= \left(\pmb{Y}_{i1}, \ldots, \pmb{Y}_{iP}\right)^T
 \end{equation}
which has dimension $\sum_{j=1}^P n_{ij}$, where $P$ is the number of periods.
\begin{align*}
    i&=1, \ldots, n \\
    j&=1,\ldots, P\\ 
    k&=1, \ldots, n_{ij} \\ 
    L&=\max_{ij}\{n_{ij}\} 
\end{align*}\nonumber
    
\end{frame}

\section{Generalized estimating equations}
\begin{frame}[fragile]{Why using GEE?}
\begin{enumerate}
    \item  Assuming that  $Y_{ijk}$ has a distribution in the exponential family,  generalized linear mixed models (GLMM) or generalized estimating equations (GEE) can be used.
    \item While the GLMM explicitly models within-subject correlation using random effects, the GEE implicitly explains such correlations using sandwich variance estimates \citep{liang1986longitudinal}.
    \item The GEE estimators yield a more solid inference about the fixed effects than the GLMM that depends on the distributional assumptions about the random effects. \citep{zhang2012new}.
\end{enumerate}
\begin{block}{GEE}    
Since the main goal of a crossover experiment is to obtain a robust inference about treatment effects, GEE models are more suitable than GLMM models.
\end{block}
   \end{frame}

\begin{frame}[fragile]{Notation}
\vfill
 \begin{block}{Model}
     \begin{equation}\nonumber
      E(Y_{ijk})=\mu_{ijk} \qquad
     g(\mu_{ijk})=\pmb{x}^T_{ijk}\pmb{\beta} =\eta_{ijk}
 \end{equation}
 \begin{equation}\nonumber
      \eta_{ijk}=\mu +\text{Period} + \text{Treatment} +\text{1st order carryover}
 \end{equation} 
\begin{equation}\nonumber
      \eta_{ijk}= \mu +\gamma_j + \tau_{d[i,j]}+ \theta_{d[i,j-1]}+\cdots+\theta_{d[i,j-P+1]} 
\end{equation}

 \end{block}
 where:
\begin{itemize}
    \item $g(\cdot)$ is the link function.
    \item $\pmb{x}_{ijk}$ is the vector of the design matrix corresponding to $k$th response from the $i$th experimental unit at the  $j$th period.
    \item $\pmb{\beta}$ is the vectors of fixed effects.
    \item $\tau_{d[i,j]}$ is the effect of treatment and   $\theta_{d[i,j-u]}$  is the carry-over effect of order $u$
\end{itemize}

\end{frame}
 \begin{frame}[fragile]{GEE}
 Because observations of the same experimental unit are correlated, parameter estimation is carried out using GEE \citep{liang1986longitudinal}.
 \begin{block}{Estimating equation for $\pmb{\beta}$}
     \begin{equation}\nonumber
        \pmb{U}(\pmb{\beta})=\left[ \left\{\sum_{i=1}^{n_i} \pmb{x}_{mi}^T \pmb{D}\left( \frac{\partial \pmb{\mu}_i}{\partial\eta}\right) [\pmb{V}(\pmb{\mu}_i)]^{-1}\left(\frac{\pmb{y}_{i}- \pmb{\mu}_i}{\phi} \right) \right\}_{m=1, \ldots, q} \right]_{q\times 1} 
     \end{equation}
 \end{block}
 where:
 \begin{itemize}
     \item $\pmb{\mu}_i=(\mu_{i11}, \ldots, \mu_{iPn_{iP}})^T$.
     \item $\pmb{y}_{i}=(y_{i11}, \ldots, y_{iPn_{iP}})^T$.
     \item $\pmb{x}_{mi}$ is the $m$th column of the design matrix of the ith experimental unit.
     \item $\phi$ is the dispersion parameter.    
     \item $n_i=\sum_{j=1}^P n_{ij}$ is the total number of observations of the $i$th experimental unit.
 \end{itemize}
\end{frame}
 \begin{frame}[fragile]{Variance component}
 \begin{block}{$\pmb{V}(\pmb{\mu}_i)$ }
     \begin{equation}\nonumber
        \pmb{V}(\pmb{\mu}_i)=\left[\pmb{D}(V(\mu_{ijk})^{\frac{1}{2}}) \pmb{R}(\pmb{\pmb{ \alpha}}) \pmb{D}({V}(\mu_{ijk})^{\frac{1}{2}})\right]_{P \times P} 
     \end{equation}
 \end{block}
in the case that $Y_{ijk}$ is binary, $V(\mu_{ijk})=\mu_{ijk}(1-\mu_{ijk})$ and $\pmb{D}(V(\mu_{ijk}))$ would be of the form:
\begin{equation}\nonumber
    \pmb{D}(V(\mu_{ijk}))=\begin{pmatrix}
     \mu_{i11}(1-\mu_{i11}) &  0 & \cdots & 0\\
     0 & \mu_{i12}(1- \mu_{i12}) & \cdots & 0 \\
     0 & 0 & \cdots & 0 \\
      \vdots & \vdots & \ddots & \vdots\\
     0&0& \cdots  &  \mu_{iPL}(1- \mu_{iPL})
    \end{pmatrix}_{PL \times PL}
\end{equation} 
\end{frame}

\begin{frame}[fragile]{Correlation structure}
$R(\pmb{ \alpha})$ is the correlation matrix related to the covariance matrix $\pmb{\Sigma}_{i}=Var(\pmb{Y}_{i})$ as follows:
 \begin{block}{Correlation }
     \begin{equation}\nonumber
        \left( \pmb{\Sigma}_{i}\right)_{PL \times PL}= \left( \pmb{D}(Var(Y_{ijk})^{\frac{1}{2}}) \pmb{R}(\pmb{\alpha}) \pmb{D}(Var(Y_{ijk})^{\frac{1}{2}})\right)_{PL \times PL}.
     \end{equation}
 \end{block} 
It is assumed that the correlation has a structure to more parsimoniously estimate $\pmb{ \alpha}$. For example:
\begin{equation}\nonumber
     \pmb{R}(\pmb{\alpha}) = \begin{pmatrix}
     1 & \alpha & \cdots& \alpha\\
     \alpha& 1 & \cdots & \alpha\\
     \vdots& \vdots & \ddots & \vdots\\
     \alpha& \alpha& \cdots & 1
    \end{pmatrix}_{PL\times PL} \;      \pmb{R}(\pmb{\alpha}) = \begin{pmatrix}
     1 & \alpha & \cdots& \alpha^{PL-1}\\
     \alpha& 1 & \cdots & \alpha^{PL-2}\\
     \vdots& \vdots & \ddots & \vdots\\
     \alpha^{PL-1}& \alpha^{PL-2}& \cdots & 1
    \end{pmatrix}_{PL\times PL}
\end{equation}
\end{frame}

\begin{frame}[fragile]{Usual GEE}
 \begin{block}{Estimating equation for $\pmb{\beta}$}
     \begin{equation}\nonumber
        \pmb{U}_1(\pmb{\beta})=\left[ \left\{\sum_{i=1}^{n_i} \pmb{x}_{mi}^T \pmb{D}\left( \frac{\partial \pmb{\mu}_i}{\partial\eta}\right) [\pmb{V}(\pmb{\mu}_i)]^{-1}\left(\frac{\pmb{y}_{i}- \pmb{\mu}_i}{\phi} \right) \right\}_{m=1, \ldots, q} \right]_{q\times 1} 
     \end{equation}
 \end{block}
 \begin{block}{Estimating equation for $\pmb{\alpha}$}
     \begin{equation}\nonumber
        \pmb{U}_2(\pmb{ \alpha})=\sum_{i=1}^S \sum_{k=1}^{n_i}\left(\frac{\partial \pmb{\varepsilon}_{ik}}{\partial \pmb{ \alpha}} \right)^t \pmb{H}_{ik}^{-1} \left(\pmb{W}_{ik} - \pmb{\varepsilon}_{ik} \right)
     \end{equation}
 \end{block}
 where $\pmb{H}_{ik}=\pmb{D}(V(r_{ijk}))_{q\times q}$ is a diagonal matrix, $\pmb{\varepsilon}_{ik}=E(\pmb{W}_{ik})_{q\times 1}$ and 
$\pmb{W}_{ik}= (r_{i1k}r_{i2k},$ $r_{i1k}r_{i3k},\ldots,r_{i(T-1)k}r_{iTk})^t_{q\times 1}$, with $r_{ijk}$ is the  $ijk$-th Pearson residual, and  $q={T \choose 2}$.
\end{frame}
\begin{frame}[fragile]{Model selection}
 The $QIC$ is defined as:
 \begin{block}{$QIC$}
     \begin{equation}\nonumber
       QIC=-2QL(\hat{\mu};\pmb{I})+2trace(\pmb{\hat{\Omega}}^{-1}_I\pmb{\hat{V}_R})
     \end{equation}
 \end{block}
where:
\begin{itemize}
    \item $\hat{\mu}_{ijk}=\hat{\eta}_{ijk}=g^{-1}(\pmb{x}_{ijk}\hat{\beta})$ is the estimated expected value of observation $Y_{ijk}$  under the model assuming the correlation matrix $R$.
    \item  $\hat{\Omega}_I$ is the estimated covariance matrix of $\pmb{\beta}$ under independence.
    \item $\hat{V}_R$ is the covariance matrix of  $\pmb{\beta}$ under the model with correlation matrix   $R(\pmb{ \alpha})$.
\end{itemize}
\begin{block}{Use}
    After fitting several models with the same formula, but with different correlation structures, the model with the lowest $QIC$ is selected because it presents the best balance between goodness-of-fit and complexity.
\end{block}
\end{frame}
\section{Deviance Analysis}

\begin{frame}[fragile]{First Result}
 \begin{block}{Theorem 1}
Let $\hat{\pmb{ \beta}}$ be the solution of the GEE and $\pmb{\beta}= (\pmb{\beta}_1^t, \pmb{ \beta}_2^t)^t$. Under the null hypothesis $H_0:\,\pmb{\beta}_1=\pmb{\beta}_{10}$, the test statistic, $T_w$, defined as:
\begin{equation}\nonumber
T_w=N (\hat{ \pmb{ \beta}}_1- \pmb{ \beta}_{10})^t V_{\hat{ \pmb{ \beta}}_1}^{-1} (\hat{ \pmb{ \beta}}_1-\pmb{ \beta}_{10})
\end{equation}
under $H_0$ and under some regularity conditions, follows an asymptotic distribution  $T_w\sim F_{T_W} $ with $F_{T_W}$ defined as:
\begin{equation}\nonumber
P(T_W\leq w)=F_{T_W}(w)=\left(\prod_{i=1}^p b_i \right)\sum_{j=0}^\infty \left( a_j \int_0^w \frac{y^{\frac{1}{2}+j-1}e^{-\frac{y}{2c_1}}}{(2c_1)^{\frac{1}{2}+j}\Gamma(\frac{1}{2}+j)} dy \right)
\end{equation}
 \end{block} 
\end{frame}


\begin{frame}[fragile]{Deviance analysis}
where $c_i$ as the $i$-th eigenvalue of the matrix  $\pmb{Q}=\pmb{Q}_0^{-1}\pmb{Q}_1$.   $\pmb{Q}_0$ and $\pmb{Q}_1$ are, respectively,
\begin{align*}
\pmb{Q}_0=&\frac{1}{n}\sum_{i=1}^S\sum_{k=1}^{n_i} \tilde{\pmb{D}}_{ik} \pmb{V}(\pmb{\mu}_i)^{-1} \pmb{D}_{ik} \\
\pmb{Q}_1=&\frac{1}{n}\sum_{i=1}^S\sum_{k=1}^{n_i}  \tilde{\pmb{D}}_{ik} \pmb{V}(\pmb{\mu}_i)^{-1}Cov(\pmb{Y}_{ik}) \pmb{V}(\pmb{\mu}_i)^{-1} \pmb{D}_{ik}\\
\pmb{Cov}(\pmb{Y}_{ik}) =& \pmb{D}\left(  \pmb{\mu}_i \right)^{\frac{1}{2}} \left(\frac{1}{n}\sum_{i=1}^S \sum_{i=1}^{n_i} \pmb{D}\left(  \pmb{\mu}_i \right)^{-\frac{1}{2}} \pmb{S}_{ik} \pmb{S}_{ik}^t  \pmb{D}\left(  \pmb{\mu}_i \right)^{-\frac{1}{2}} \right) \pmb{D}\left(  \pmb{\mu}_i \right)^{\frac{1}{2}}\\
b_i =& \sqrt{ \frac{c_1}{c_i}}\\
A(c_i, r)=&\left[ \prod_{k=0}^{r-1}\left(\frac{1}{2}+k \right)\left(1-\frac{c_1}{c_r} \right)^{r}\right] \left(r! \right)^{-1}
\end{align*}
\end{frame}
\begin{frame}[fragile]{Sketch of proof}
\begin{enumerate}
    \item By Theorem 2 of \cite{liang1986longitudinal} it is known that  $\hat{\pmb{\beta}}$ converges in probability to $\pmb{\beta}$, furthermore:
\begin{equation}\nonumber
\sqrt{n} (\hat{\pmb{\beta}}-\pmb{\beta}) \stackrel{D}{\rightarrow} N(\pmb{0}, Var(\hat{\pmb{\beta}}))
\end{equation} 
\item A robust and consistent estimator for the variance of  $\hat{\pmb{\beta}}$ \citep{panv} is given by:
\begin{align}
\widehat{Var}(\hat{\pmb{\beta}})=&\left(\sum_{i=1}^n \pmb{D}_{i}^t\pmb{V}(\pmb{\mu}_i)^{-1}\pmb{D}_{i} \right)^{-1} 
\left(\sum_{k=1}^n \pmb{D}_{i}^t \pmb{V}(\pmb{\mu}_i)^{-1} \widehat{Cov}(\pmb{y}_{i}) \pmb{V}(\pmb{\mu}_i)^{-1}\pmb{D}_{i} \right)\nonumber\\ &\left(\sum_{k=1}^n \pmb{D}_{i}^t \pmb{V}(\pmb{\mu}_i)^{-1}\pmb{D}_{i} \right)\nonumber\\
&= \pmb{A}^{-1}\pmb{B}\pmb{A}\nonumber
\end{align}
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Sketch of proof}
\begin{itemize}
    \item[3.] For the Result 1 of \cite{rk90}, it is obtained that the $T_w$ statistic:
\begin{equation}
T_w =n (\hat{\pmb{\beta}}_1- \pmb{\beta}_{10})^t V_{\hat{\pmb{ \beta}}_1}^{-1} (\hat{\pmb{\beta}}_1- \pmb{\beta}_{10}) \stackrel{n\rightarrow \infty}{\sim} \sum_{j=1}^p c_j \chi_j^2 \nonumber    
\end{equation}
\item[4.] Then, the distribution of $W=\sum_{j=1}^p c_j \chi_j^2$ is found based on \cite{raro} as follows:
\begin{align}
F_W(w)&=P(W\leq w)=\left(\prod_{i=1}^p b_i \right)\sum_{j=0}^\infty \left( a_j \int_0^w g_j(y)dy \right)\nonumber\\
&=\left(\prod_{i=1}^p b_i \right)\sum_{j=0}^\infty \left( a_j \int_0^w \frac{y^{\frac{1}{2}+j-1}e^{-\frac{y}{2c_1}}}{(2c_1)^{\frac{1}{2}+j}{ \Gamma}(\frac{1}{2}+j)} dy \right)\nonumber \square
\end{align}
\end{itemize}

\end{frame}

\section{Kronecker correlation structure}
\begin{frame}[fragile]{New correlation structure}
We propose a correlation structure of the form:
 \begin{block}{Correlation }
\begin{equation}\nonumber
    \pmb{R}(\pmb{\alpha})=\pmb{\Psi}\otimes\pmb{R}_1(\pmb{\alpha}_1)
\end{equation}
 \end{block} 
 where $\pmb{R}_1(\pmb{\alpha}_1)$  is the within-period correlation matrix, $\pmb{\Psi}$ is the between-period correlation matrix, and $\otimes$ represents the Kronecker product \citep{harville}. If $P=3$ y $L=10$, an example of $\pmb{R}(\pmb{\alpha})$ would be:
\begin{equation}\nonumber
    \pmb{R}_1(\pmb{\alpha}_1) = \begin{pmatrix}
     1 & \alpha_1 & \alpha_1&\ldots & \alpha_1\\
     \alpha_1 & 1 & \alpha_1 &\ldots & \alpha_1\\
     \alpha_1 & \alpha_1 & 1 &\ldots & \alpha_1\\
     \vdots & \vdots & \vdots & \ddots & \vdots\\
     \alpha_1 & \alpha_1 & \ldots &\alpha_1 & 1
    \end{pmatrix}_{10\times 10}, \; \pmb{\Psi}=\begin{pmatrix}
     1 & \psi_{12} & \psi_{13}\\
     \psi_{12} & 1  & \psi_{23}\\
     \psi_{13} &\psi_{23} & 1
    \end{pmatrix}_{3\times 3}
\end{equation}
\end{frame}
\begin{frame}[fragile]{Kronecker GEE}
 \begin{block}{Estimating equation for $\pmb{\beta}$}
     \begin{equation}\nonumber
        \pmb{U}_1(\pmb{\beta})=\left[ \left\{\sum_{i=1}^{n_i} \pmb{x}_{mi}^T \pmb{D}\left( \frac{\partial \pmb{\mu}_i}{\partial\eta}\right) [\pmb{V}(\pmb{\mu}_i)]^{-1}\left(\frac{\pmb{y}_{i}- \pmb{\mu}_i}{\phi} \right) \right\}_{m=1, \ldots, q} \right]_{q\times 1} 
     \end{equation}
 \end{block}
 \begin{block}{Estimating equation for $\pmb{\alpha}_1$}
     \begin{equation}\nonumber
        \pmb{U}_2(\pmb{ \alpha}_1)=\sum_{i=1}^n \left(\frac{\partial \pmb{\varepsilon}_{i}}{\partial \pmb{ \alpha}_1} \right)^T \pmb{H}_{i}^{-1} \left(\pmb{W}_{i} - \pmb{\varepsilon}_{i} \right)
     \end{equation}
 \end{block}
 \end{frame}

 \begin{frame}[fragile]{Estimating equation for $\pmb{\Psi}$}
 \begin{block}{ $\pmb{\Psi}=\left(\psi_{jj'}\right)_{P\times P}$}
 \begin{equation}\nonumber
\hat{\psi}_{jj'}=\frac{1}{n} \sum_{i=1}^n tr\left(\pmb{R}_1(\pmb{\hat{\alpha}}_1)^{-1}(\pmb{r}_{(j)i}-\bar{\pmb{r}}_{(j)})(\pmb{r}_{(j')i}-\bar{\pmb{r}}_{(j')})^T\right)
\end{equation}
 \end{block}
 where:
 \begin{itemize}
     \item $\pmb{r}_{(j)i}$ is the vector of Pearson residuals for the $i$th experimental unit at the $j$th  period.
     \item $\bar{\pmb{r}}_{(j)}$ is the average (over i),  $j\neq j'$, with $j= 1, \ldots, P$.
 \end{itemize}
\begin{block}{Steps}
\begin{enumerate}
    \item  $\pmb{R}(\pmb{\alpha}_1)$ is estimated with the residuals grouped by experimental unit with Equation $U_2$
    \item With the residuals grouped by period and the experimental unit, the $\pmb{\Psi}$ matrix is estimated with $\hat{\psi}_{jj'}$, but discounting the correlation already captured by $\pmb{R}(\pmb{\alpha}_1)$.    
\end{enumerate}
\end{block} 
 \end{frame}

\begin{frame}[fragile]{Second result}
\vfill
 \begin{block}{Theorem 2}
The estimator of $\pmb{R}(\pmb{\alpha})$ given by $\pmb{\hat{\Psi}}\otimes\pmb{R}_1(\pmb{\hat{\alpha}}_1)$ is asymptotically unbiased and consistent. 
 \end{block}
 
\end{frame}

\begin{frame}[fragile]{Sketch of proof}
\begin{enumerate}
    \item Define the theoretical Pearson residuals of response of the $k$th of the $i$th experimental unit at the $j$th period as:
\begin{equation}\nonumber
    R_{ijk}=\frac{Y_{ijk}-\mu_{ijk}}{{\phi}\sqrt{V(\mu_{ijk})})}
\end{equation}
\item Adapting the results of \cite{cox1968general} and \cite{cordeiro2004Pearson} it is true that:
\begin{equation}\nonumber
    R_{ijk}\xrightarrow[n\to \infty]{d} N(0,1)
\end{equation}
\item If the matrix  $\pmb{R}=(\pmb{r}_1,\ldots, \pmb{r}_n)_{P\times nL}$ with $\pmb{r}_i$ of residuals of the $i$th individual is defined as:
\begin{equation}\nonumber
      \pmb{r}_i=\begin{pmatrix}
    r_{i11}& r_{i21}& \cdots& r_{i1L}\\
    r_{i21}& r_{i22}& \cdots &r_{i2L}\\
    \vdots& \vdots& \ddots& \vdots\\
    r_{iP1}& r_{i2P}& \cdots &r_{iPL}\\
    \end{pmatrix}
    \end{equation}
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Sketch of proof}
\begin{itemize}
    \item[4.] If the matrix $\pmb{\Gamma}$ is defined as:
{\scriptsize
    \begin{equation}\nonumber
    \pmb{\Gamma}^T=\begin{pmatrix}
     \frac{\pmb{1}}{\sqrt{n}} &\vdots &      \pmb{G}
    \end{pmatrix}^T=\begin{pmatrix}
    \frac{1}{\sqrt{n}} & \frac{1}{\sqrt{n}}& \frac{1}{\sqrt{n}}& \frac{1}{\sqrt{n}} & \cdots&\frac{1}{\sqrt{n}}\\
    \frac{1}{\sqrt{2}} & -\frac{1}{\sqrt{2}}& 0& 0 &\cdots& 0\\
    \frac{1}{\sqrt{6}} & \frac{1}{\sqrt{6}}& -\frac{2}{\sqrt{6}}& 0 & \cdots&0\\
    \frac{1}{\sqrt{12}} & \frac{1}{\sqrt{12}}& \frac{1}{\sqrt{12}}& -\frac{3}{\sqrt{12}} & \cdots&0\\
    \vdots & \vdots& \vdots& \vdots & \ddots&\vdots\\
    \frac{1}{\sqrt{(n(n-1))}} & \frac{1}{\sqrt{n(n-1)}}& \frac{1}{\sqrt{n(n-1)}}& \frac{1}{\sqrt{n(n-1)}} & \cdots&-\frac{n-1}{\sqrt{n(n-1)}}\\
    \end{pmatrix}
\end{equation}}
\item[5.] Adapting the results of \citep{srivastava2008models}: 
\begin{equation}\nonumber
\pmb{R}(\pmb{\Gamma}\otimes \pmb{I}_L)=\begin{pmatrix}
\sqrt{n}\bar{\pmb{r}} &\vdots & \pmb{R}(\pmb{G}\otimes \pmb{I}_L)
\end{pmatrix} 
\end{equation}
\begin{equation}\nonumber
    \pmb{R}\left(\pmb{I}_n\otimes\pmb{ \Psi}^{-1} \right)\pmb{R}^T=n\bar{\pmb{r}}\otimes \pmb{ \Psi}^{-1} \pmb{R}+\pmb{Z}(\pmb{G}^T\otimes\pmb{I}_L )\pmb{Z}^T
\end{equation}
\end{itemize}
\end{frame}


\begin{frame}[fragile]{Sketch of proof}
\begin{itemize}
    \item[6.] By properties of $R_{ijk}$ and the central limit theorem, we get:
    \begin{align}\nonumber
    E(\pmb{Z}_i) =\pmb{0}_{P\times L} \, \; \text{and }
    Var(Vec(\pmb{Z}_i))=\pmb{ \Psi} \otimes \pmb{R}_1(\pmb{\alpha}_1))
\end{align}
\item[7.]
\begin{equation}\nonumber
    E(\pmb{z}_{(j)1}, \pmb{z}^T_{(j)1})=\Psi_{jj}\pmb{R}_1(\pmb{\alpha}_1) \qquad E(\pmb{Z}_1, \pmb{Z}^T_1)=(trace(\pmb{\Psi}))\pmb{R}_1(\pmb{\alpha}_1)\nonumber
\end{equation}
\item[8.] Therefore, since $\pmb{\Psi}_{jj}=1$, $\forall j=1,\ldots, P$, $trace(\pmb{\Psi})=1$ we get:
\begin{equation}\nonumber
   E\left( \pmb{R}_1(\pmb{\alpha}_1)^{\frac{1}{2}}\pmb{z}_{(j')k} \pmb{z}^T_{(j)k} \pmb{R}_1(\pmb{\alpha}_1)^{\frac{1}{2}}\right)=\pmb{\Psi}_{jj'}\pmb{I}_P, \qquad \forall k=1, \ldots, n-1
\end{equation}
\item[9.] Futhermore, 
\begin{equation}\nonumber
    Cov[\pmb{R}_1(\pmb{\alpha}_1)^{\frac{1}{2}} \pmb{z}_{(k)j} \pmb{z}^T_{(i)j} \pmb{R}_1(\pmb{\alpha}_1)^{\frac{1}{2}}, \pmb{R}_1(\pmb{\alpha}_1)^{\frac{1}{2}} \pmb{z}_{(k)j'} \pmb{z}^T_{(i)j'} \pmb{R}_1(\pmb{\alpha}_1)^{\frac{1}{2}} ]=\pmb{0}   
\end{equation}
\end{itemize}
$\square$
\end{frame}

\section{Semiparametric modelling}
\begin{frame}[fragile]{Semiparametric GEE}
We propose a semiparametric model with B-splines for the time and carry-over effects as follows:
 \begin{block}{Correlation }
\begin{align*}
     E(Y_{ijk})&=\mu_{ijk}, \qquad Var(Y_{ijk})=\phi V(\mu_{ijk})\nonumber \\
     g(\mu_{ijk})&=\pmb{x}_{ijk}^T\pmb{\beta} + f(\pmb{Z}_{ijk})+ \sum_{c=1}^C f_c(\pmb{Z}_{ijk})\\
     \pmb{V}(\pmb{\mu}_i)&=\left[\pmb{D}(V(\mu_{ijk})^{\frac{1}{2}}) \pmb{R}(\pmb{\pmb{ \alpha}}) \pmb{D}({V}(\mu_{ijk})^{\frac{1}{2}})\right]_{ L\times L}
\end{align*}
 \end{block} 
 where $\pmb{Z}_{ijk}$ is the time at which the $k$-th observation of the $i$-th experimental unit was measured in the $j$-th period.  $f$  is a function describing the time’s effect period, $f_c$ is a function describing the previous treatment carry-over effect on the current period, with $f_c(\pmb{Z}_{i1k})=0$.
\end{frame}


\begin{frame}[fragile]{Carry-overs}
For example in the blood pressure data set described in Table \ref{table100},  $\pmb{Z}_{i11}=-30, \; \pmb{Z}_{i12}=-15, \ldots, \pmb{Z}_{i1\,10}=240$, that is, the minutes of measurement within each period
\begin{block}{Simple carry-over}
     $C=2$, that is, $f_1$  is the carry-over effect of A, $f_2$  is the carry-over effect of B. 
\end{block}
\begin{block}{Complex carry-over}
     $C=6$, that is, $f_1$  is the carry-over effect of A on B, $f_2$  is the carry-over effect of A on C,  $f_3$ is the carry-over effect of B on A,  $f_4$ is the carry-over effect of B on C,  $f_5$ is the carry-over effect of C on A and  $f_6$ is the carry-over effect of C on B
\end{block}
\begin{block}{Advantages}
The usual GEE modelling does not allow modelling complex carry-over effects, but this methodology does allow modelling them.
\end{block}
\end{frame}
\begin{frame}[fragile]{GEE splines}
Let $\{s_1(t), \ldots, s_{m}(t)\}$ be a basis splines, then  the  $f$ and $f_c$ functions can be approximated through the following equations \citep{yu2012spline}, respectively,
\begin{align*}
    \hat{f}(t)= \sum_{b=1}^m \hat{\alpha}_{1b}s_b(t),  \,
    \hat{f}_c(t)= \sum_{b=1}^m \hat{\alpha}_{cb}s_b(t), \forall c=1, \ldots,C
\end{align*}
the following generalized estimation equations are proposed for $\alpha_{1}=\{\alpha_{11},\ldots, \alpha_{1m}\}$, $ \alpha_c=\{\alpha_{c1},\ldots, \alpha_{cm}\}$,  $\pmb{\beta}$ y $\pmb{R}(\pmb{\alpha})$:
\begin{block}{For the time effect}
\scriptsize
     \begin{equation}\nonumber
        U_1(\pmb{\alpha_1}, t|\pmb{\beta}, \pmb{\alpha}_2, \pmb{\alpha})=\sum_{i=1}^n \left\{ diag\left(\frac{\partial \mu_{ijk}}{\partial \pmb{\alpha}_1}\right)\right\}_i \frac {\pmb{V}^{-1}_{1i}}{\phi} \left(\pmb{y}_i-\pmb{\mu}_i\left[\pmb{X}_i \pmb{\beta},\sum_{b=1}^m \alpha_{1b}s_b(t)  ,\hat{f}_c(\pmb{Z}_{i}) \right]\right) 
     \end{equation}
 \end{block}
\end{frame}

\begin{frame}[fragile]{GEE splines}

\begin{block}{For the carry-over effect}
\scriptsize
     \begin{equation}\nonumber
       U_2(\pmb{\alpha_c}, t|\pmb{\beta}, \pmb{\alpha}_1, \pmb{\alpha})=\sum_{i=1}^n \left\{ diag\left(\frac{\partial \mu_{ijk}}{\partial \pmb{\alpha}_2}\right)\right\}_i\frac {\pmb{V}^{-1}_{2i}}{\phi} \left(\pmb{y}_i-\pmb{\mu}_i\left(\pmb{X}_i \pmb{\beta},\sum_{b=1}^m \alpha_{2b}s_b(t)  ,\hat{f}(\pmb{Z}_{1i}) \right)\right)
     \end{equation}
 \end{block}
 \begin{block}{For $\pmb{\beta}$ and correlation matrix}
     \begin{equation}\nonumber \scriptsize
      U_3(\pmb{\beta}|\pmb{\alpha}_1, \pmb{\alpha}_2, \pmb{\alpha})=\sum_{i=1}^n \left\{  diag\left(\frac{\partial \mu_{ijk}}{\partial \pmb{\beta}}\right)\right\}_i\frac {\pmb{V}^{-1}_{3i}}{\phi} \left(\pmb{y}_i-\pmb{\mu}_i\left[\pmb{X}_i \pmb{\beta},\hat{f}(\pmb{Z}_{i}) ,\hat{f}_c(\pmb{Z}_{i}) \right]\right)
     \end{equation}
     \begin{equation}\nonumber
       U_4(\pmb{ \alpha}|\pmb{\beta}, \pmb{\alpha}_1, \pmb{\alpha}_2)=\sum_{i=1}^S \sum_{k=1}^{n_i}\left(\frac{\partial \pmb{\varepsilon}_{ik}}{\partial \pmb{ \alpha}} \right)^T \pmb{F}_{ik}^{-1} \left(\pmb{W}_{ik} - \pmb{\varepsilon}_{ik} \right)
     \end{equation}
 \end{block}
\end{frame}

\begin{frame}[fragile]{Steps of estimation extending the Fisher Scoring}
\small
\begin{enumerate}
    \item Set initial values  $\pmb{\alpha}_{2}^{(0)}$, $\pmb{\beta}^{(0)}$ and $\pmb{\alpha}^{(0)}$
    \item Find the value $\pmb{\alpha}_{1}^{(1)}$ that solves the equation
    \[U_1(\pmb{\alpha_1}^{(1)}, t|\pmb{\beta}^{(0)}, \pmb{\alpha}_2^{(0)}, \pmb{\alpha}^{(0)})=0\]  
 \item Find the value $\pmb{\alpha}_{2}^{(1)}$ that solves the equation
 \[U_2(\pmb{\alpha_2}^{(1)}, t|\pmb{\beta}^{(0)}, \pmb{\alpha}_1^{(1)}, \pmb{\alpha}^{(0)})=0\]  
 \item Find the value $\pmb{\beta}^{(1)}$ that solves the equation
 \[U_3(\pmb{\beta}^{(1)}|\pmb{\alpha}_1^{(1)}, \pmb{\alpha}_2^{(1)}, \pmb{\alpha}^{(0)})=0\]  
  \item Find the value $\pmb{\alpha}^{(1)}$ that solves the equation
 \[U_4(\pmb{\alpha}^{(1)}|\pmb{\beta}^{(1)},\pmb{\alpha}_1^{(1)}, \pmb{\alpha}_2^{(1)}, \pmb{\alpha}^{(0)})=0\]
 \item Repeat steps (2) to (5) until convergence.
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Estimate for $\pmb{\beta}$}
If solving the equation $U_3$ with the Fisher scoring algorithm, in the  $m$-th step, the estimator of $\pmb{\beta}$ is given by:
\begin{align}
 \pmb{\beta}^{(m+1)}&=\pmb{\beta}^{(m)}- \left\{ \sum_{i=1}^n \pmb{X}_i^T \pmb{W}_i^{(m)} \pmb{X}_i \right\}^{-1} \left\{\sum_{i=1}^n\pmb{X}_i^T \pmb{W}_i^{(m)} \left(\pmb{N}_i^{(m)} \right)^{-1}  \pmb{u}_i^{(m)}  \right\} \nonumber
 \end{align}
\begin{block}{$\hat{\pmb{\beta}}$ is like a weighted least squares estimator}
   \begin{equation}\nonumber
    \pmb{\beta}^{(m+1)}=\left\{ \sum_{i=1}^n \pmb{X}_i^T \pmb{W}_i^{(m)} \pmb{X}_i \right\}^{-1} \left\{\sum_{i=1}^n\pmb{X}_i^T \pmb{W}_i^{(m)}  \pmb{z}_i^{(m)}  \right\} 
\end{equation}
\end{block}
where
 \begin{align*}
     \pmb{z}^{(m)}_i&=\pmb{X}_i\pmb{\beta}^{(m)}-\pmb{N}_i^{-1(m)}\pmb{u}_i^{(m)} 
\end{align*}
\end{frame}
\begin{frame}[fragile]{Estimate for $\pmb{\beta}$}
where
\begin{align*}
\pmb{W}_i^{(m)}&=\left\{  diag\left(\frac{\partial \mu_{ijk}^{(m)}}{\partial \pmb{\beta}^{(m)}}\right)\right\}_i\pmb{V}^{-1}_{3i} \left\{  diag\left(\frac{\partial \mu_{ijk}^{(m)}}{\partial \pmb{\beta}^{(m)}}\right)\right\}_i ^{-1} \\
\pmb{u}_i^{(m)}&=\left(\pmb{y}_i-\pmb{\mu}_i^{(m)}\left[\pmb{X}_i \pmb{\beta}^{(m)},\hat{f}(\pmb{Z}_{i}^{(m)} ,\hat{f}_c(\pmb{Z}_{i}^{(m)}) \right]\right)\\
\pmb{N}_i^{(m)}&=\left\{  diag\left(\frac{\partial \mu_{ijk}^{(m)}}{\partial \pmb{\beta}^{(m)}}\right)\right\}_i\\
 \pmb{A}&=\sum_{i=1}^n  \pmb{N}_i \pmb{V}^{-1}_{3i}\pmb{N}_i^T \\
   \pmb{B}&=\sum_{i=1}^n  \pmb{N}_i \pmb{V}^{-1}_{3i}\left(\pmb{y}_i-\pmb{\hat{\mu}}_i\right)\left(\pmb{y}_i-\pmb{\hat{\mu}}_i\right)^T \pmb{V}^{-1}_{3i} \pmb{N}_i^T
\end{align*}
\end{frame}

\begin{frame}[fragile]{Third result}
 \begin{block}{Theorem 3}
Under the assumption that the $r$-th  derivative of $f$ and $f_c$ is bounded for some $r\geq 2$ and that the number of knots $m=m_n\rightarrow \infty$, but $\frac{m}{n}\rightarrow 0$ then $\hat{\pmb{\beta}}-\pmb{\beta} \xrightarrow{n\rightarrow\infty}\pmb{0}$. Also, if $m=O\left( n^{\frac{1}{(2^r+1)}}\right)$ then:
\begin{align}
   & \frac{1}{n}\sum_{i=1}^n \sum_{j=1}^{n_i} \left\{\sum_{b=1}^m \hat{\alpha}_{1b}s_b(\pmb{Z}_{ijk})- f(\pmb{Z}_{ijk})\right\}^2 = O\left(n^{-\frac{2r}{(2r+1)}}\right) \nonumber\\
   & \frac{1}{n}\sum_{i=1}^n \sum_{j=1}^{n_i} \left\{\sum_{b=1}^m \hat{\alpha}_{cb}s_b(\pmb{Z}_{ijk})- f_c(\pmb{Z}_{ijk})\right\}^2 = O\left(n^{-\frac{2r}{(2r+1)}}\right)\nonumber\\
  &  \sqrt{n}(\hat{\pmb{\beta}}-\pmb{\beta}) \rightarrow N(0, \pmb{A}^{-1}\pmb{B}\pmb{A}^{-1})\nonumber
\end{align}
 \end{block} 
\end{frame}

\begin{frame}[fragile]{Sketch of proof}
\begin{enumerate}
    \item We assume that there exists a random variables, $\delta_{ijk}$, with $E(\delta_{ijk})=0$ and $Var(\delta_{ijk})\leq\infty$ and continuous functions $g_1, \ldots, g_m$  such that:
\begin{equation}
    x_{ijkl}=g_l(\pmb{Z}_{ijk})+ \delta_{ijkl} \nonumber
\end{equation}
\item Since $Y_{ijk}$ belongs to  the exponential family and by \cite{liang1986longitudinal}, and \cite{lehmann2006theory}, then:
\begin{align}
         &E(U_1(\pmb{\epsilon}_i, t))=0 \, \forall t\in \mathbb{R}\nonumber\\
         &0 < E\left(U_1(\pmb{\epsilon}_i, t) U_1(\pmb{\epsilon}_i, t)^T\right) < \infty \nonumber\\
         &E(U_2(\pmb{\epsilon}_i, t))=0, \qquad 0< E\left(U_2(\pmb{\epsilon}_i, t)U_2(\pmb{\epsilon}_i, t)^T\right) <\infty, \; \forall t\in \mathbb{R}\nonumber\\
        &E(U_3(\pmb{\epsilon}_i, t))=0, \qquad 0< E\left(U_3(\pmb{\epsilon}_i, t)U_3(\pmb{\epsilon}_i, t)^T\right) <\infty, \; \forall t\in \mathbb{R}\nonumber\\
        &E(U_4(\pmb{\epsilon}_i))=0, \qquad 0< E\left(U_4(\pmb{\epsilon}_i)U_4(\pmb{\epsilon}_i)^T\right) <\infty \nonumber
\end{align}
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Sketch of proof}
\begin{itemize}
    \item[3.] when $s\to 0 $, the following properties hold, respectively:
\begin{align}
    \sup_{ijk}\left| E(U_1(\pmb{\epsilon}_{ijk}+s, t))-b_{1ijk}s\right|&=O(s^2), \; \forall t\in \mathbb{R},\nonumber \\
     \sup_{ijk}\left| E(U_2(\pmb{\epsilon}_{ijk}+s, t))-b_{2ijk}s\right|&=O(s^2), \; \forall t\in \mathbb{R},\nonumber \\
     \sup_{ijk}\left| E(U_3(\pmb{\epsilon}_{ijk}+s))-b_{3ijk}s\right|&=O(s^2)\nonumber,
\end{align}
\item[4.] When $s\to 0$, there exist constants $c>0$ and,  $C<\infty$ such that:
\begin{align}
    & \sup_{ijk}\left\{ E(U_1(\pmb{\epsilon}_{ijk}+s, t)-U_1(\pmb{\epsilon}_{ijk}, t))^2\right\} \leq C|s|, \; \forall t\in \mathbb{R}\nonumber\\
     &\sup_{ijk}\left\{ E(U_2(\pmb{\epsilon}_{ijk}+s, t)-U_2(\pmb{\epsilon}_{ijk}, t))^2\right\} \leq C|s|, \; \forall t\in \mathbb{R}\nonumber\\
     &\sup_{ijk}\left\{ E(U_3(\pmb{\epsilon}_{ijk}+s)-U_1(\pmb{\epsilon}_{ijk}))^2\right\} \leq C|s|\nonumber
\end{align}
\item[4.] And using Theorem 1 and Theorem 2 of \cite{he2002estimation}, the theorem are obtained. $\square$
\end{itemize}

\section{Conclusions}
\begin{frame}[fragile]{Conclusions}
\vfill
    \begin{enumerate}
\item Using GEE, unbiased estimates of important design effects are obtained, regardless of the presence or absence of carryover effects, in addition to obtaining a deviance analysis table for design analysis.
       \item The family of correlation structures built using the Kronecker product allows greater flexibility in the correlation structure and helps to better analyze crossover designs.
       \item The semiparametric methodology provides an estimate of the effects over time, and reduces the bias of the resulting estimators. It also allows making asymptotic inferences and better modelling temporary carryover behaviors that would be intractable in parametric scenarios.
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Future Research}
\vfill
    \begin{enumerate}
\item  Check the theoretical and practical performance of the combination of the Kronecker structure with the semiparametric modelling (In progress).
\item Build a semi-parametric model that allows you to estimate complex carryover effects (In progress).
\item Propose a methodology to deal with missing data, either due to random missing data or non-ignorable dropout.
\item Implement a Bayesian approach to work out carryover effects and possible treatment interactions over time.

\end{enumerate}
\end{frame}
