10  Métodos basados en árboles


En este capítulo utilizaremos árboles de decisión, Random Forest y Gradient Boosting para predecir el peso y clasificar la salud de los animales. Estos métodos pueden representar relaciones no lineales e interacciones sin especificarlas manualmente, pero requieren control de complejidad y una evaluación rigurosa fuera de muestra.

10.2 Árbol de regresión


Código 
receta_arbol_peso <- recipe(
  peso_kg ~
    consumo_kg_dia + consumo_agua_l_dia +
    edad_inicio_meses + sexo + raza + tiempo +
    temperatura_C + humedad_pct + precipitacion_mm +
    actividad_h_dia + sistema_produccion + zona,
  data = train_arboles
) |>
  step_unknown(all_nominal_predictors()) |>
  step_novel(all_nominal_predictors())

modelo_arbol_reg <- decision_tree(
  cost_complexity = tune(),
  tree_depth = tune(),
  min_n = tune()
) |>
  set_engine("rpart") |>
  set_mode("regression")

flujo_arbol_reg <- workflow() |>
  add_recipe(receta_arbol_peso) |>
  add_model(modelo_arbol_reg)


Código 
rejilla_arbol <- grid_random(
  cost_complexity(range = c(-5, -1)),
  tree_depth(range = c(2L, 12L)),
  min_n(range = c(5L, 50L)),
  size = 10
)

folds_arboles_rapidos <- vfold_cv(
  train_arboles,
  v = 3
)

set.seed(2026)
resultados_arbol_reg <- tune_grid(
  flujo_arbol_reg,
  resamples = folds_arboles_rapidos,
  grid = rejilla_arbol,
  metrics = metric_set(rmse),
  control = control_grid(verbose = TRUE)
)

show_best(resultados_arbol_reg, metric = "rmse", n = 5)

10.3 Random Forest para regresión


Random Forest combina muchos árboles construidos con muestras y subconjuntos aleatorios de predictores. El promedio de los árboles reduce la inestabilidad de un árbol individual.


Código 
receta_forest_peso <- receta_arbol_peso |>
  step_dummy(all_nominal_predictors()) |>
  step_zv(all_predictors())

modelo_rf_reg <- rand_forest(
  mtry = tune(),
  min_n = tune(),
  trees = 100
) |>
  set_engine(
    "ranger",
    importance = "permutation"
  ) |>
  set_mode("regression")

flujo_rf_reg <- workflow() |>
  add_recipe(receta_forest_peso) |>
  add_model(modelo_rf_reg)

rejilla_rf <- grid_space_filling(
  finalize(mtry(), train_arboles),
  min_n(range = c(2L, 40L)),
  size = 20
)


Código 
rejilla_rf <- grid_random(
  mtry(range = c(2L, 12L)),
  min_n(range = c(5L, 50L)),
  size = 10
)

set.seed(2026)

  flujo_rf_reg,
  resamples = folds_arboles_rapidos,
  grid = rejilla_rf,

La importancia por permutación mide cuánto empeora la predicción cuando se altera una variable. No demuestra causalidad y puede repartirse entre predictores altamente correlacionados.


10.4 Gradient Boosting para regresión


Gradient Boosting construye árboles secuencialmente. Cada nuevo árbol intenta corregir errores del conjunto anterior. La tasa de aprendizaje, la profundidad, el número de árboles y la regularización controlan la complejidad.


Código 
receta_boost <- receta_forest_peso

modelo_boost <- boost_tree(
  trees = tune(),
  tree_depth = tune(),
  learn_rate = tune(),
  loss_reduction = tune(),
  min_n = tune(),
  sample_size = tune(),
  mtry = tune()
) |>
  set_engine("xgboost") |>
  set_mode("regression")

flujo_boost <- workflow() |>
  add_recipe(receta_boost) |>
  add_model(modelo_boost)


Código 
parametros_boost <- parameters(
  trees(range = c(30L, 50L)),
  tree_depth(range = c(2L, 3L)),
  learn_rate(range = c(-2, -1)),
  loss_reduction(),
  min_n(range = c(5L, 15L)),
  sample_prop(range = c(0.8, 1.0)),
  finalize(mtry(), train_arboles)
)

set.seed(2026)
rejilla_boost <- grid_latin_hypercube(
  parametros_boost,
  size = 3
)

set.seed(2026)
resultados_boost <- tune_grid(
  flujo_boost,
  resamples = vfold_cv(train_arboles, v = 2),
  grid = rejilla_boost,
  metrics = metric_set(rmse),
  control = control_grid(verbose = TRUE)
)

show_best(resultados_boost, metric = "rmse", n = 3)

10.6 Árbol y Random Forest para clasificación


Código 
receta_clas_arbol <- recipe(
  salud_alta_ml ~
    consumo_kg_dia + consumo_agua_l_dia +
    edad_inicio_meses + sexo + raza + tiempo +
    temperatura_C + humedad_pct + precipitacion_mm +
    actividad_h_dia + sistema_produccion + zona,
  data = train_arboles
) |>
  step_unknown(all_nominal_predictors()) |>
  step_novel(all_nominal_predictors()) |>
  step_dummy(all_nominal_predictors()) |>
  step_zv(all_predictors())

modelo_rf_clas <- rand_forest(
  mtry = tune(),
  min_n = tune(),
  trees = 100
) |>
  set_engine(
    "ranger",
    importance = "permutation",
    probability = TRUE
  ) |>
  set_mode("classification")

flujo_rf_clas <- workflow() |>
  add_recipe(receta_clas_arbol) |>
  add_model(modelo_rf_clas)

set.seed(2026)
